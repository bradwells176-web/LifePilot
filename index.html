export default async function handler(req, res){ if(req.method !== "GET"){ return res.status(405).json({ error:"Method not allowed" }); } const apiKey = process.env.ELEVENLABS_API_KEY; if(!apiKey){ return res.status(500).json({ error: "ELEVENLABS_API_KEY is not configured." }); } try{ const url = new URL( "https://api.elevenlabs.io/v2/voices" ); url.searchParams.set( "language", "en" ); url.searchParams.set( "use_cases", "conversational" ); url.searchParams.set( "include_total_count", "true" ); const response = await fetch( url.toString(), { headers:{ "xi-api-key":apiKey } } ); if(!response.ok){ const error = await response.text(); console.error(error); return res.status( response.status ).json({ error:"Unable to load voices." }); } const data = await response.json(); /* Never send American voices to LifePilot's UI. */ const voices = (data.voices || []) .filter(voice=>{ const accent = String( voice.labels?.accent || "" ).toLowerCase(); const gender = String( voice.labels?.gender || "" ).toLowerCase(); const american = accent.includes("american") || accent.includes("usa") || accent.includes("us"); const allowed = accent.includes("british") || accent.includes("australian") || accent.includes("canadian"); const genderAllowed = gender.includes("male") || gender.includes("female"); return ( !american && allowed && genderAllowed ); }) .map(voice=>({ voice_id: voice.voice_id, name: voice.name, labels: voice.labels || {}, preview_url: voice.preview_url || "" })); return res.status(200).json({ voices }); }catch(error){ console.error(error); return res.status(500).json({ error:"Unable to load LifePilot voices." }); }
